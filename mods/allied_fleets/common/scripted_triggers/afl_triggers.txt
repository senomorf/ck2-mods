# Allied Fleets scripted triggers. A trigger here is the one place a condition lives: the
# decisions and events use it, and so do the in-game tests (tests/game/allied_fleets/).
# Pattern: vanilla/common/scripted_triggers/00_scripted_triggers.txt, called as `NAME = yes`.
# Constructs: has_mercenary_band (vanilla/decisions/ze_decisions.txt), has_raised_standing_troops in
# title scope (vanilla/common/scripted_triggers/00_scripted_triggers.txt; box probe: yes while a
# company is hired), band_creator (vanilla/events/ze_mercenary_events.txt);
# any_war with any_attacker and any_defender (vanilla/events/HFP_crusade_events.txt,
# .cwtools/scopes.cwt), war_with = event_target (vanilla/events/jd_artifact_events.txt),
# has_earmarked_regiments (this mod's tests), any_demesne_title with tier, has_building and
# holding_type in title scope (box probe 6, docs/ck2-notes.md).

# afl_in_contract_trigger: lends or borrows ships now, has a request or a charter offer out or in,
# or has founded a galley company. One contract per ruler at a time, so two allies can't trade
# fleets to avoid paying for their own, and one request per lender at a time, so two askers can't
# overwrite each other's terms; a founder's ships are the company's.
# Scope: a character.
afl_in_contract_trigger = {
	OR = {
		has_character_flag = afl_company
		has_character_flag = afl_borrowing
		has_character_flag = afl_lending
		has_character_flag = afl_request_pending
		has_character_flag = afl_request_incoming
		has_character_flag = afl_offer_out
		has_character_flag = afl_offer_in
	}
}

# afl_war_partner_trigger: the scoped ruler (the lender) and FROM (the ruler asking) are allied,
# or on the same side of a war. Crusade partners are war allies only, which is_allied_with misses.
# Scope: the lender; FROM: the ruler asking (afl_ask_war_loan).
afl_war_partner_trigger = {
	OR = {
		is_allied_with = FROM
		any_war = {
			OR = {
				AND = {
					any_attacker = { character = ROOT }
					any_attacker = { character = FROM }
				}
				AND = {
					any_defender = { character = ROOT }
					any_defender = { character = FROM }
				}
			}
		}
	}
}

# afl_can_borrow_trigger: may this ruler ask for a war loan now?
# Scope: the ruler asking.
afl_can_borrow_trigger = {
	war = yes
	NOT = { afl_in_contract_trigger = yes }
}

# afl_can_charter_trigger: may this ruler hire ships by charter now? Any landed ruler, at war or at
# peace, outside another contract (owner, 2026-11-14: allies, war partners and anyone friendly).
# Scope: the ruler hiring.
afl_can_charter_trigger = {
	is_landed = yes
	NOT = { afl_in_contract_trigger = yes }
}

# afl_can_pay_lump_trigger: can the ruler hiring pay the sum on signing for afl_ask galleys now? Two
# months' wages, 0.6 gold a galley (owner, 2026-11-15; afl_set_charter_terms_effect). `wealth`
# takes a number, so each count has its line. Scope: the ruler hiring, with afl_ask set.
afl_can_pay_lump_trigger = {
	OR = {
		AND = { is_variable_equal = { which = afl_ask value = 50 } wealth = 30 }
		AND = { is_variable_equal = { which = afl_ask value = 100 } wealth = 60 }
		AND = { is_variable_equal = { which = afl_ask value = 150 } wealth = 90 }
		AND = { is_variable_equal = { which = afl_ask value = 200 } wealth = 120 }
		AND = { is_variable_equal = { which = afl_ask value = 300 } wealth = 180 }
		AND = { is_variable_equal = { which = afl_ask value = 400 } wealth = 240 }
		AND = { is_variable_equal = { which = afl_ask value = 500 } wealth = 300 }
	}
}

# afl_charter_over_trigger: has the charter ended early? When the two are at war with each other,
# when the lender's title has no landed holder, or when the fleet is gone. The term (twelve
# monthly checks after signing) and unpaid wages end it in
# afl_charter_check_effect. Scope: the borrower; event_target:afl_lender = the title's holder.
afl_charter_over_trigger = {
	OR = {
		war_with = event_target:afl_lender
		event_target:afl_lender = {
			OR = {
				is_alive = no
				is_landed = no
			}
		}
		NOT = { has_earmarked_regiments = afl_charter }
	}
}

# afl_has_ships_trigger: a holding whose buildings give galleys. Every vanilla building with
# `galleys` (vanilla/common/buildings/: 00_buildings.txt, 00_Castle.txt, 00_tribal.txt,
# 00_nomad.txt) upgrades from its level 1, so level 1 stands for the chain; a family palace has
# galleys = 25 of its own (vanilla/common/static_modifiers.txt). afl_count_base_effect lists the
# same buildings. Scope: a title.
afl_has_ships_trigger = {
	tier = BARON
	OR = {
		has_building = ct_shipyard_1
		has_building = ct_rep_arsenal_1
		has_building = ca_shipyard_1
		has_building = tp_shipyard_1
		has_building = fp_shipyard_1
		has_building = tb_shipyard_1
		has_building = no_harbor_1
		holding_type = family_palace
	}
}

# afl_can_lend_trigger: may this ruler lend ships now? It lends from its own holdings only (owner,
# 2026-11-15), so one of them needs ships; how many it can spare is counted when it is asked
# (afl_count_base_effect). Scope: the lender.
afl_can_lend_trigger = {
	any_demesne_title = { afl_has_ships_trigger = yes }
	NOT = { afl_in_contract_trigger = yes }
}

# afl_war_loan_over_trigger: has the war loan ended? At peace (the fleet has disband_on_peace),
# when the fleet is gone, when the two are at war with each other, when the lender recalls the
# ships, or when the lender's title has no landed holder.
# Scope: the borrower; event_target:afl_lender = the lender's title's holder.
afl_war_loan_over_trigger = {
	OR = {
		war = no
		NOT = { has_earmarked_regiments = afl_war_loan }
		war_with = event_target:afl_lender
		event_target:afl_lender = {
			OR = {
				is_alive = no
				is_landed = no
				has_character_flag = afl_recalling
			}
		}
	}
}

# Galley companies (step 3). A ruler founds a mercenary company of its own galleys, which anyone in
# reach hires through the game's Mercenaries list (owner, 2026-11-15; README.md, "Galley companies").

# afl_can_found_trigger: may this ruler found a galley company now? It needs a holding of its own
# with ships, no other contract, and no mercenary company already (has_mercenary_band also sees
# a company founded by vanilla's own decision). Scope: the ruler.
afl_can_found_trigger = {
	is_landed = yes
	afl_can_lend_trigger = yes
	NOT = { has_mercenary_band = yes }
}

# afl_company_trigger: a galley company founded by this mod. Scope: a title.
afl_company_trigger = {
	mercenary = yes
	has_title_flag = afl_company
}

# afl_company_idle_trigger: a galley company nobody has hired now (its fleet is raised only while
# hired). Scope: the company's title.
afl_company_idle_trigger = {
	has_raised_standing_troops = no
}
