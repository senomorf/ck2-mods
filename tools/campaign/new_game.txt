# tools/campaign.sh, the vanilla run: a new 1066 game running in observe mode at full speed.
# Coordinates are for the 1280x720 Xvfb screen (the game runs full screen at 1280x720) and were read
# off the screenshots of PR #15's probe runs on the fleet runner (2026-11-13; "run N" below) and of
# the live test box's probes (2026-11-13; "box"). Waits are on game.log lines where the game writes
# one; the fixed waits left are short margins.
#
# -startgame -start_date goes straight to the 1066 character selection, skipping the main menu,
# Single Player, New Game and the era screen (box: the selection showed about 30 s after the
# history marker, against about 140 s of waits through the menus).
launch -startgame -start_date=1066.9.15
until 300 After first history execution
# The lobby's dice button picks a random playable character; game.log gets "Human ... set as
# primary local" once it does (box). Clicks before the lobby is up land on a loading screen.
clickuntil 995 686 300 set as primary local
shot lobby
wait 3
click 1163 654      # "Play" (run 7): the Game Rules window was up 2 s later (box)
wait 5
clickuntil 937 601 120 Launching SINGLEPLAYER-game   # Game Rules: "Start Game" (run 8)
# The game loads under the Game Rules window, then starts paused under the "Welcome to Crusader
# Kings II!" splash (common/hints.txt splash_screen, then splash_screen_2), even with hints=no in
# settings.txt (run 10); box: the splash came about 27 s after the Launching line. new_game_observe.txt
# closes it and types `observe` in the console; observing writes "Human ... unset as primary local".
wait 15
repeat 300 unset as primary local new_game_observe.txt
shot ingame
click 1100 50       # the date: a click unpauses (run 12)
wait 5
click 1240 50       # "+" speed, four times to the fastest (run 12)
wait 2
click 1240 50
wait 2
click 1240 50
wait 2
click 1240 50
wait 2
move 640 600        # off the buttons, so no tooltip covers the date
wait 10
shot running
info
