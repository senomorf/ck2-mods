# tools/campaign.sh, each mod's run: load the save (the only one in "save games") straight into the
# game, then run it at full speed. Coordinates: see new_game.txt.
#
# -startgame -start_save=<file in save games> skips the main menu and Single Player > Continue: on
# the live test box (2026-11-13) game.log said "Launching SINGLEPLAYER-game" 34 s after the start,
# and the map showed, paused, 5 s later, still observing ("Country: No Character").
launch -startgame -start_save=%SAVE%
until 300 Launching SINGLEPLAYER-game
wait 10
shot loaded
# The date: a click unpauses. Clicked again until the pixels of the date and the speed sign
# (1016,43, 204x16) keep changing with no red in them (the sign is red while paused), since a click
# while the map still loads is lost (run 43531833583).
clickchange 1100 50 180 1016 43 204 16
click 1240 50       # "+" speed, four times
wait 2
click 1240 50
wait 2
click 1240 50
wait 2
click 1240 50
wait 2
move 640 600
wait 10
shot running
info
