# tools/campaign.sh --steps probe: load a save and run a file of effects through the console, to try
# a mod's events or effects in about a minute instead of a campaign run. Use with
#   CAMPAIGN_PROBE_SAVE=<a .ck2>  CAMPAIGN_PROBE_EFFECTS=<effects file>  [CAMPAIGN_PROBE_MOD=<mod>]
# The console's `run FILE` reads FILE relative to the user folder (CK2_USER_DIR; on Windows
# Documents/Paradox Interactive/Crusader Kings II), and runs it with no player character in observe
# mode; `log = "..."` lines in it reach game.log (live test box, 2026-11-13). An absolute path does
# nothing, with no message.
launch -startgame -start_save=%SAVE%
until 300 Launching SINGLEPLAYER-game
wait 10
focus
key grave
wait 3
type run ck2test_effects.txt
key Return
wait 5
shot ran
key grave
click 1100 50       # unpause, so events the file fired get their days
wait 30
shot later
