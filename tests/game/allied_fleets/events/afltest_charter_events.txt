namespace = afltest

# In-game tests for allied_fleets' charters (README.md, "Charter"), run by tools/campaign.sh
# through tools/harness.py after the war loan's (afltest_events.txt): afltest.12 starts them.
# Generated; every check is written as a pass line and a fail line.
# Part 1 (afltest.30): a charter's steps on chosen rulers, checked at each step.
# Part 2 (afltest.32): AI rulers ask AI lenders for charters as a player would, and AI answers;
# each is checked when its ships should have sailed (afltest.33), then ended.
# Part 3 (afltest.34): requests and offers whose ruler choosing the count has afl_base forced,
# so every count and every "never mind" is chosen; checked 35 days on (afltest.35, afltest.36).
# afltest.38 (from afltest.13, near the year's end): a charter whose ships sail before the
# year's last save, for save_checks.txt.
# Constructs as in afltest_events.txt; export_to_variable with wealth (this mod's war loan tests).

# afltest.30: pick the rulers, check the conditions and the terms, then a 50-galley charter through
# its signing, sailing, margin, warning and term, a change of the lender's title's holder, unpaid
# wages, and a borrower's death. Event targets: afltest_cb, a free ruler at peace (hires);
# afltest_cl, a free ruler whose own holdings can spare 50 galleys, not at war with afltest_cb
# (lends); afltest_cp, another free ruler (stands in as an old lender).
character_event = {
	id = afltest.30
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		clear_event_target = afl_borrower
		clear_event_target = afl_lender
		clear_event_target = afl_lender_title
		clear_event_target = afl_port
		any_independent_ruler = {
			limit = { is_landed = yes afl_can_lend_trigger = yes }
			afl_count_base_effect = yes
		}
		random_independent_ruler = {
			limit = {
				is_landed = yes
				war = no
				afl_can_charter_trigger = yes
				NOT = { has_character_flag = afl_test_requester }
				any_independent_ruler = {
					is_landed = yes
					afl_can_lend_trigger = yes
					check_variable = { which = afl_base value = 50 }
					NOT = { character = PREV }
					NOT = { war_with = PREV }
				}
			}
			save_event_target_as = afltest_cb
			set_character_flag = afl_test_requester
			random_independent_ruler = {
				limit = {
					is_landed = yes
					afl_can_lend_trigger = yes
					check_variable = { which = afl_base value = 50 }
					NOT = { character = PREV }
					NOT = { war_with = PREV }
				}
				save_event_target_as = afltest_cl
			}
		}
		random_independent_ruler = {
			limit = {
				is_landed = yes
				NOT = { afl_in_contract_trigger = yes }
				NOT = { character = event_target:afltest_cb }
				NOT = { character = event_target:afltest_cl }
			}
			save_event_target_as = afltest_cp
		}
		if = {
			limit = {
				OR = {
					NOT = { event_target:afltest_cb = { always = yes } }
					NOT = { event_target:afltest_cl = { always = yes } }
					NOT = { event_target:afltest_cp = { always = yes } }
				}
			}
			log = "CK2TEST fail charter scenario setup: missing a free ruler at peace with a lender who can spare 50 galleys"
		}
		if = {
			limit = {
				event_target:afltest_cb = { always = yes }
				event_target:afltest_cl = { always = yes }
				event_target:afltest_cp = { always = yes }
			}
			log = "CK2TEST pass charter scenario setup"
			log = "afltest: charter scenario borrower [afltest_cb.GetBestName] ([afltest_cb.GetID]), lender [afltest_cl.GetBestName] ([afltest_cl.GetID]), stand-in [afltest_cp.GetBestName] ([afltest_cp.GetID])"
			event_target:afltest_cb = {
				while = { count = 1000 limit = { wealth = 0 } wealth = -25 }
				while = { count = 30 limit = { NOT = { wealth = 0 } } wealth = 1 }
				wealth = 500
			}

			# The conditions (common/scripted_triggers/afl_triggers.txt).
			if = { limit = { event_target:afltest_cb = { afl_can_charter_trigger = yes NOT = { afl_can_borrow_trigger = yes } } } log = "CK2TEST pass a ruler at peace may hire ships by charter" }
			if = { limit = { NOT = { event_target:afltest_cb = { afl_can_charter_trigger = yes NOT = { afl_can_borrow_trigger = yes } } } } log = "CK2TEST fail a ruler at peace may hire ships by charter: [afltest_cb.GetBestName] ([afltest_cb.GetID]) may not, or may borrow for a war" }
			event_target:afltest_cb = { set_character_flag = afl_offer_in }
			if = { limit = { event_target:afltest_cb = { NOT = { afl_can_charter_trigger = yes } } } log = "CK2TEST pass an offer out blocks another contract" }
			if = { limit = { NOT = { event_target:afltest_cb = { NOT = { afl_can_charter_trigger = yes } } } } log = "CK2TEST fail an offer out blocks another contract: it does not" }
			event_target:afltest_cb = { clr_character_flag = afl_offer_in }

			# The sum on signing (afl_can_pay_lump_trigger): 30 gold for 50 galleys, 300 for 500.
			event_target:afltest_cb = { set_variable = { which = afl_ask value = 500 } }
			if = { limit = { event_target:afltest_cb = { afl_can_pay_lump_trigger = yes } } log = "CK2TEST pass a treasury of 500 pays the sum on signing for 500 galleys" }
			if = { limit = { NOT = { event_target:afltest_cb = { afl_can_pay_lump_trigger = yes } } } log = "CK2TEST fail a treasury of 500 pays the sum on signing for 500 galleys: it does not" }
			event_target:afltest_cb = {
				while = { count = 1000 limit = { wealth = 0 } wealth = -25 }
				while = { count = 30 limit = { NOT = { wealth = 0 } } wealth = 1 }
				wealth = 20
				set_variable = { which = afl_ask value = 50 }
			}
			if = { limit = { event_target:afltest_cb = { NOT = { afl_can_pay_lump_trigger = yes } } } log = "CK2TEST pass a treasury of 20 can't pay the sum on signing for 50 galleys" }
			if = { limit = { NOT = { event_target:afltest_cb = { NOT = { afl_can_pay_lump_trigger = yes } } } } log = "CK2TEST fail a treasury of 20 can't pay the sum on signing for 50 galleys: it can" }
			event_target:afltest_cb = {
				while = { count = 1000 limit = { wealth = 0 } wealth = -25 }
				while = { count = 30 limit = { NOT = { wealth = 0 } } wealth = 1 }
				wealth = 500
			}

			# A 50-galley charter, signed.
			event_target:afltest_cb = {
				save_event_target_as = afl_borrower
				set_character_flag = afl_request_pending
				set_character_flag = afl_charter
				set_variable = { which = afl_ask value = 50 }
			}
			event_target:afltest_cl = { save_event_target_as = afl_lender set_character_flag = afl_request_incoming }
			event_target:afltest_cb = { afl_set_charter_terms_effect = yes }
			event_target:afltest_cb = { export_to_variable = { which = global_afltest_b_before value = wealth } }
			event_target:afltest_cl = { export_to_variable = { which = global_afltest_l_before value = wealth } }
			event_target:afltest_cl = { afl_start_charter_effect = yes }
			if = { limit = { event_target:afltest_cb = { is_variable_equal = { which = afl_lump value = 30 } is_variable_equal = { which = afl_margin value = 3.75 } is_variable_equal = { which = afl_wages value = 15 } } event_target:afltest_cl = { is_variable_equal = { which = afl_lump value = 30 } is_variable_equal = { which = afl_margin value = 3.75 } is_variable_equal = { which = afl_wages value = 15 } } } log = "CK2TEST pass the terms for 50 galleys: 30 gold on signing, 3.75 a month, wages about 15, on both" }
			if = { limit = { NOT = { event_target:afltest_cb = { is_variable_equal = { which = afl_lump value = 30 } is_variable_equal = { which = afl_margin value = 3.75 } is_variable_equal = { which = afl_wages value = 15 } } event_target:afltest_cl = { is_variable_equal = { which = afl_lump value = 30 } is_variable_equal = { which = afl_margin value = 3.75 } is_variable_equal = { which = afl_wages value = 15 } } } } log = "CK2TEST fail the terms for 50 galleys: 30 gold on signing, 3.75 a month, wages about 15, on both: [afltest_cb.afl_lump.GetValue], [afltest_cb.afl_margin.GetValue], [afltest_cb.afl_wages.GetValue] and [afltest_cl.afl_lump.GetValue], [afltest_cl.afl_margin.GetValue], [afltest_cl.afl_wages.GetValue]" }
			if = { limit = { event_target:afl_port = { port = yes } } log = "CK2TEST pass the ships gather in a port of the lender's" }
			if = { limit = { NOT = { event_target:afl_port = { port = yes } } } log = "CK2TEST fail the ships gather in a port of the lender's: [afl_port.GetName] has no port" }
			event_target:afltest_cb = { export_to_variable = { which = global_afltest_b_after value = wealth } }
			event_target:afltest_cl = { export_to_variable = { which = global_afltest_l_after value = wealth } }
			set_variable = { which = global_afltest_paid which = global_afltest_b_before }
			subtract_variable = { which = global_afltest_paid which = global_afltest_b_after }
			set_variable = { which = global_afltest_got which = global_afltest_l_after }
			subtract_variable = { which = global_afltest_got which = global_afltest_l_before }
			if = { limit = { check_variable = { which = global_afltest_paid value = 29.999 } NOT = { check_variable = { which = global_afltest_paid value = 30.001 } } check_variable = { which = global_afltest_got value = 29.999 } NOT = { check_variable = { which = global_afltest_got value = 30.001 } } } log = "CK2TEST pass signing pays the lender 30 gold" }
			if = { limit = { NOT = { check_variable = { which = global_afltest_paid value = 29.999 } NOT = { check_variable = { which = global_afltest_paid value = 30.001 } } check_variable = { which = global_afltest_got value = 29.999 } NOT = { check_variable = { which = global_afltest_got value = 30.001 } } } } log = "CK2TEST fail signing pays the lender 30 gold: the borrower paid [global_afltest_paid.GetValue], the lender got [global_afltest_got.GetValue], not 30" }
			if = { limit = { event_target:afltest_cb = { has_character_flag = afl_borrowing has_character_flag = afl_charter } event_target:afltest_cl = { has_character_flag = afl_lending has_character_flag = afl_charter } event_target:afltest_cb = { NOT = { has_character_flag = afl_request_pending } } event_target:afltest_cl = { NOT = { has_character_flag = afl_request_incoming } } } log = "CK2TEST pass a charter marks both rulers" }
			if = { limit = { NOT = { event_target:afltest_cb = { has_character_flag = afl_borrowing has_character_flag = afl_charter } event_target:afltest_cl = { has_character_flag = afl_lending has_character_flag = afl_charter } event_target:afltest_cb = { NOT = { has_character_flag = afl_request_pending } } event_target:afltest_cl = { NOT = { has_character_flag = afl_request_incoming } } } } log = "CK2TEST fail a charter marks both rulers: flags not as expected" }
			if = { limit = { event_target:afltest_cb = { NOT = { has_earmarked_regiments = afl_charter } NOT = { has_character_flag = afl_charter_sailed } } } log = "CK2TEST pass the ships are not there before they gather" }
			if = { limit = { NOT = { event_target:afltest_cb = { NOT = { has_earmarked_regiments = afl_charter } NOT = { has_character_flag = afl_charter_sailed } } } } log = "CK2TEST fail the ships are not there before they gather: [afltest_cb.GetBestName] ([afltest_cb.GetID]) already has them" }
			if = { limit = { event_target:afltest_cl = { any_demesne_title = { afl_has_ships_trigger = yes OR = { has_holding_modifier = afl_ships_away_10 has_holding_modifier = afl_ships_away_1 } } } } log = "CK2TEST pass signing puts Ships away on the lender's holdings while the ships gather" }
			if = { limit = { NOT = { event_target:afltest_cl = { any_demesne_title = { afl_has_ships_trigger = yes OR = { has_holding_modifier = afl_ships_away_10 has_holding_modifier = afl_ships_away_1 } } } } } log = "CK2TEST fail signing puts Ships away on the lender's holdings while the ships gather: none carries it" }
			if = { limit = { event_target:afltest_cl = { NOT = { any_vassal = { any_demesne_title = { OR = { has_holding_modifier = afl_ships_away_10 has_holding_modifier = afl_ships_away_1 } } } } } } log = "CK2TEST pass the lender's vassals keep their ships under a charter" }
			if = { limit = { NOT = { event_target:afltest_cl = { NOT = { any_vassal = { any_demesne_title = { OR = { has_holding_modifier = afl_ships_away_10 has_holding_modifier = afl_ships_away_1 } } } } } } } log = "CK2TEST fail the lender's vassals keep their ships under a charter: a vassal's holding carries the cut" }

			# The first monthly check, 30 days on: the ships sail; no margin yet.
			event_target:afltest_cb = { export_to_variable = { which = global_afltest_b_before value = wealth } }
			event_target:afltest_cl = { export_to_variable = { which = global_afltest_l_before value = wealth } }

			if = { limit = { NOT = { event_target:afltest_cb = { has_character_flag = afl_borrowing has_character_flag = afl_charter } event_target:afltest_cl = { has_character_flag = afl_lending has_character_flag = afl_charter } event_target:afltest_cb = { has_character_flag = afl_charter_sailed has_earmarked_regiments = afl_charter is_variable_equal = { which = afl_months value = 0 } } event_target:afltest_cl = { any_demesne_title = { afl_has_ships_trigger = yes OR = { has_holding_modifier = afl_ships_away_10 has_holding_modifier = afl_ships_away_1 } } } } } log = "CK2TEST fail the first check sails the ships at the lender's port: no fleet, or the count of months is [afltest_cb.afl_months.GetValue]" }

			if = { limit = { NOT = { event_target:afltest_cb = { has_character_flag = afl_borrowing has_character_flag = afl_charter } event_target:afltest_cl = { has_character_flag = afl_lending has_character_flag = afl_charter } event_target:afltest_cb = { war = no has_earmarked_regiments = afl_charter is_variable_equal = { which = afl_months value = 1 } } } } log = "CK2TEST fail peace does not end a charter, and the month counts: it ended, or the count of months is [afltest_cb.afl_months.GetValue]" }
			event_target:afltest_cb = { export_to_variable = { which = global_afltest_b_after value = wealth } }

# afltest.38 (from afltest.13, near the year's end): a 50-galley charter, signed through its steps,
# whose ships sail 30 days on, before the year's last save (save_checks.txt).
character_event = {
	id = afltest.38
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		clear_event_target = afl_borrower
		clear_event_target = afl_lender
		clear_event_target = afl_lender_title
		clear_event_target = afl_port
		any_independent_ruler = {
			limit = { is_landed = yes afl_can_lend_trigger = yes }
			afl_count_base_effect = yes
		}
		random_independent_ruler = {
			limit = {
				afl_can_charter_trigger = yes
				any_independent_ruler = {
					is_landed = yes
					afl_can_lend_trigger = yes
					check_variable = { which = afl_base value = 50 }
					NOT = { character = PREV }
					NOT = { war_with = PREV }
				}
			}
			save_event_target_as = afl_borrower
			wealth = 100
			set_character_flag = afl_request_pending
			set_character_flag = afl_charter
			set_variable = { which = afl_ask value = 50 }
			random_independent_ruler = {
				limit = {
					is_landed = yes
					afl_can_lend_trigger = yes
					check_variable = { which = afl_base value = 50 }
					NOT = { character = PREV }
					NOT = { war_with = PREV }
				}
				save_event_target_as = afl_lender
				set_character_flag = afl_request_incoming
			}
			afl_set_charter_terms_effect = yes
			event_target:afl_lender = { afl_start_charter_effect = yes }
		}
		if = {
			limit = { any_independent_ruler = { has_character_flag = afl_borrowing has_character_flag = afl_charter } }
			log = "CK2TEST pass a late charter runs into the year's last save"
		}
		if = {
			limit = { NOT = { any_independent_ruler = { has_character_flag = afl_borrowing has_character_flag = afl_charter } } }
			log = "CK2TEST fail a late charter runs into the year's last save: no pair of rulers could start one"
		}
	}
}
