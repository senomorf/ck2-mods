namespace = afltest

# In-game tests for allied_fleets (the war loan, README.md), run by tools/campaign.sh through
# tools/harness.py. Generated, with common/scripted_triggers/afltest_triggers.txt; every check
# is written as a pass line and a fail line. The mod run loads a vanilla save in observe mode, so
# nobody takes the decisions (ai = no). Part 1 runs the decision's effect for AI rulers 30 times
# and lets the AI answer afl.10 and afl.11; then 27 more (afltest.6) set the asker's
# afl_base to each count in turn, since an AI asks for the largest count offered and few rulers
# can spare 100 galleys in 1067. So every option comes up. Each loan is checked and ended
# five days on, so the few rulers who can spare 50 galleys lend again. Part 2 (afltest.10 on)
# runs each step on chosen rulers and checks it. Part 3 (afltest.13) leaves loans running into
# the year's last save, for save_checks.txt.
# Constructs, all in vanilla 3.3.5.1 or this mod: on_startup (fires for every character on load,
# .cwtools/on_actions.cwt), set_global_flag, random_independent_ruler and any_independent_ruler,
# change_variable, check_variable and is_variable_equal, while with count and limit
# (vanilla/events/game_rule_events.txt), export_to_variable value = wealth
# (vanilla/common/scripted_effects/01_scripted_effects_duels.txt), set_variable,
# subtract_variable and multiply_variable with which (.cwtools/variables_event_targets.cwt),
# global_ variables and [global_<name>.GetValue], [This.<name>.GetValue] (vanilla/localisation),
# any_war with end_war = whitepeace (vanilla/events/HF_sway_events.txt), death with death_reason
# (vanilla/events, death_accident_drowning), has_holding_modifier and
# has_instances_of_holding_modifier (box probe 6, docs/ck2-notes.md).

# afltest.1: the first ruler on_startup reaches starts the requests, once.
character_event = {
	id = afltest.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ruler = yes
		NOT = { has_global_flag = afl_test_started }
	}

	immediate = {
		set_global_flag = afl_test_started
		log = "CK2TEST start allied_fleets"
		character_event = { id = afltest.2 days = 1 }
	}
}

# afltest.2: one request from an AI ruler at war, then the next one a day later; after the
# 30th, the requests for each count (afltest.6). Two requests in three go to a ruler whose own holdings can spare 50
# galleys, if one is free; the third, and any for which none is, to any ruler with ships.
character_event = {
	id = afltest.2
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		# A delayed event inherits its chain's event targets, so clear the last run's.
		clear_event_target = afltest_asker
		clear_event_target = afl_lender
		change_variable = { which = afl_test_runs value = 1 }
		change_variable = { which = afl_test_third value = 1 }
		any_independent_ruler = {
			limit = { is_landed = yes afl_can_lend_trigger = yes }
			afl_count_base_effect = yes
		}
		random_independent_ruler = {
			limit = {
				is_landed = yes
				wealth = 0
				afl_can_borrow_trigger = yes
				NOT = { has_character_flag = afl_test_requester }
			}
			save_event_target_as = afltest_asker
		}
		if = {
			limit = {
				event_target:afltest_asker = { always = yes }
				NOT = { check_variable = { which = afl_test_third value = 3 } }
			}
			random_independent_ruler = {
				limit = {
					is_landed = yes
					afl_can_lend_trigger = yes
					check_variable = { which = afl_base value = 50 }
					NOT = { character = event_target:afltest_asker }
					NOT = { war_with = event_target:afltest_asker }
				}
				save_event_target_as = afl_lender
			}
		}
		if = {
			limit = { check_variable = { which = afl_test_third value = 3 } }
			set_variable = { which = afl_test_third value = 0 }
		}
		if = {
			limit = {
				event_target:afltest_asker = { always = yes }
				NOT = { event_target:afl_lender = { always = yes } }
			}
			random_independent_ruler = {
				limit = {
					is_landed = yes
					afl_can_lend_trigger = yes
					NOT = { character = event_target:afltest_asker }
					NOT = { war_with = event_target:afltest_asker }
				}
				save_event_target_as = afl_lender
			}
		}
		if = {
			limit = {
				event_target:afltest_asker = { always = yes }
				event_target:afl_lender = { always = yes }
			}
			event_target:afltest_asker = {
				set_character_flag = afl_test_requester
				set_character_flag = afl_test_awaiting_check
				# The decision's own effect: it saves afl_lender, then runs this for the ruler asking.
				afl_ask_war_loan_effect = yes
				character_event = { id = afltest.3 days = 5 }
			}
		}
		if = {
			limit = { NOT = { check_variable = { which = afl_test_runs value = 30 } } }
			character_event = { id = afltest.2 days = 1 }
		}
		if = {
			limit = { check_variable = { which = afl_test_runs value = 30 } }
			character_event = { id = afltest.6 days = 1 }
		}
	}
}

# afltest.3: five days after a request, for the ruler who asked. A loan is checked, then ended.
character_event = {
	id = afltest.3
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		clr_character_flag = afl_test_awaiting_check
		if = { limit = { NOT = { has_character_flag = afl_request_pending } event_target:afl_lender = { NOT = { has_character_flag = afl_request_incoming } } } log = "CK2TEST pass afl.10 and afl.11 answer the request" }
		if = { limit = { NAND = { NOT = { has_character_flag = afl_request_pending } event_target:afl_lender = { NOT = { has_character_flag = afl_request_incoming } } } } log = "CK2TEST fail afl.10 and afl.11 answer the request: a request flag is still set five days later on [Root.GetBestName] ([Root.GetID]) or [afl_lender.GetBestName] ([afl_lender.GetID])" }
if = {
	limit = { has_character_flag = afl_borrowing }
	set_variable = { which = afltest_spare which = afl_base }
	subtract_variable = { which = afltest_spare which = afl_ask }
			if = { limit = { has_earmarked_regiments = afl_war_loan } log = "CK2TEST pass an AI war loan gives the borrower a fleet" }
			if = { limit = { NOT = { has_earmarked_regiments = afl_war_loan } } log = "CK2TEST fail an AI war loan gives the borrower a fleet: [Root.GetBestName] ([Root.GetID]) has none" }
			if = { limit = { check_variable = { which = afl_ask value = 50 } NOT = { check_variable = { which = afl_ask value = 501 } } check_variable = { which = afltest_spare value = 0 } } log = "CK2TEST pass an AI request asks for 50 to 500 galleys, and no more than the lender can spare" }
			if = { limit = { NAND = { check_variable = { which = afl_ask value = 50 } NOT = { check_variable = { which = afl_ask value = 501 } } check_variable = { which = afltest_spare value = 0 } } } log = "CK2TEST fail an AI request asks for 50 to 500 galleys, and no more than the lender can spare: [Root.GetBestName] ([Root.GetID]) asked for [This.afl_ask.GetValue] of [This.afl_base.GetValue]" }
			if = { limit = { event_target:afl_lender = { any_demesne_title = { OR = { has_holding_modifier = afl_ships_away_10 has_holding_modifier = afl_ships_away_1 } } } } log = "CK2TEST pass an AI war loan puts Ships away on the lender's own holdings" }
			if = { limit = { NOT = { event_target:afl_lender = { any_demesne_title = { OR = { has_holding_modifier = afl_ships_away_10 has_holding_modifier = afl_ships_away_1 } } } } } log = "CK2TEST fail an AI war loan puts Ships away on the lender's own holdings: none on [afl_lender.GetBestName] ([afl_lender.GetID])" }
			afl_end_war_loan_effect = yes
			if = { limit = { NOT = { has_earmarked_regiments = afl_war_loan } event_target:afl_lender = { NOT = { has_character_flag = afl_lending } NOT = { any_demesne_title = { OR = { has_holding_modifier = afl_ships_away_10 has_holding_modifier = afl_ships_away_1 } } } } } log = "CK2TEST pass ending an AI war loan takes Ships away off the lender's holdings" }
			if = { limit = { NAND = { NOT = { has_earmarked_regiments = afl_war_loan } event_target:afl_lender = { NOT = { has_character_flag = afl_lending } NOT = { any_demesne_title = { OR = { has_holding_modifier = afl_ships_away_10 has_holding_modifier = afl_ships_away_1 } } } } } } log = "CK2TEST fail ending an AI war loan takes Ships away off the lender's holdings: the fleet or a modifier of [afl_lender.GetBestName] ([afl_lender.GetID]) is still there" }
		}
	}
}

# afltest.6: a request whose asker is offered every count up to one, 50 to 500 in turn, with
# 200 gold; then two offered up to 600, one with 20 to 45 gold, who withdraws on the second
# page (afl.14), and one with 65 to 90, who withdraws on the third (afl.15). Three rounds, a
# day apart; then the scenario (afltest.4). It runs afl_ask_war_loan_effect's steps, with
# afl_base set by the test instead of counted.
character_event = {
	id = afltest.6
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		clear_event_target = afltest_asker
		clear_event_target = afl_lender
		change_variable = { which = afl_test_synth value = 1 }
		change_variable = { which = afl_test_ci value = 1 }
		if = {
			limit = { check_variable = { which = afl_test_ci value = 10 } }
			set_variable = { which = afl_test_ci value = 1 }
		}
		random_independent_ruler = {
			limit = {
				is_landed = yes
				wealth = 0
				afl_can_borrow_trigger = yes
				# not one whose last request's check is still to come: that check reads its own lender
				NOT = { has_character_flag = afl_test_awaiting_check }
			}
			save_event_target_as = afltest_asker
			random_independent_ruler = {
				limit = {
					is_landed = yes
					afl_can_lend_trigger = yes
					NOT = { character = PREV }
					NOT = { war_with = PREV }
				}
				save_event_target_as = afl_lender
			}
		}
		if = {
			limit = {
				event_target:afltest_asker = { always = yes }
				event_target:afl_lender = { always = yes }
			}
			event_target:afltest_asker = {
				set_character_flag = afl_test_requester
				set_character_flag = afl_test_awaiting_check
				save_event_target_as = afl_borrower
				set_character_flag = afl_request_pending
				event_target:afl_lender = { set_character_flag = afl_request_incoming }
				if = { limit = { ROOT = { is_variable_equal = { which = afl_test_ci value = 1 } } } set_variable = { which = afl_base value = 50 } wealth = 200 }
				if = { limit = { ROOT = { is_variable_equal = { which = afl_test_ci value = 2 } } } set_variable = { which = afl_base value = 100 } wealth = 200 }
				if = { limit = { ROOT = { is_variable_equal = { which = afl_test_ci value = 3 } } } set_variable = { which = afl_base value = 150 } wealth = 200 }
				if = { limit = { ROOT = { is_variable_equal = { which = afl_test_ci value = 4 } } } set_variable = { which = afl_base value = 200 } wealth = 200 }
				if = { limit = { ROOT = { is_variable_equal = { which = afl_test_ci value = 5 } } } set_variable = { which = afl_base value = 300 } wealth = 200 }
				if = { limit = { ROOT = { is_variable_equal = { which = afl_test_ci value = 6 } } } set_variable = { which = afl_base value = 400 } wealth = 200 }
				if = { limit = { ROOT = { is_variable_equal = { which = afl_test_ci value = 7 } } } set_variable = { which = afl_base value = 500 } wealth = 200 }
				if = { limit = { ROOT = { is_variable_equal = { which = afl_test_ci value = 8 } } } set_variable = { which = afl_base value = 600 } while = { count = 400 limit = { wealth = 0 } wealth = -25 } wealth = 45 }
				if = { limit = { ROOT = { is_variable_equal = { which = afl_test_ci value = 9 } } } set_variable = { which = afl_base value = 600 } while = { count = 400 limit = { wealth = 0 } wealth = -25 } wealth = 90 }
				log = "afltest: [This.GetBestName] ([This.GetID]) is offered up to [This.afl_base.GetValue] galleys by [afl_lender.GetBestName] ([afl_lender.GetID])"
				character_event = { id = afl.10 }
				character_event = { id = afltest.3 days = 5 }
			}
		}
		if = {
			limit = { NOT = { check_variable = { which = afl_test_synth value = 27 } } }
			character_event = { id = afltest.6 days = 1 }
		}
		if = {
			limit = { check_variable = { which = afl_test_synth value = 27 } }
			character_event = { id = afltest.4 days = 15 }
		}
	}
}

# Scenario: the war loan's steps on chosen rulers. Event targets: afltest_b, a ruler at war
# (borrows); afltest_l, a ruler whose own holdings can spare 50 galleys (lends); afltest_p, a
# ruler at peace; afltest_n, a ruler with no holding with ships; afltest_b2, a second ruler at
# war (borrows, then dies).

# afltest.10: pick the rulers, check the conditions and the arithmetic, then a 50-galley loan
# through two months, a change of the lender's title's holder and two months unpaid, then a loan
# and a recall, then a loan and peace (afltest.11).
character_event = {
	id = afltest.10
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		any_independent_ruler = {
			limit = { is_landed = yes afl_can_lend_trigger = yes }
			afl_count_base_effect = yes
		}
		random_independent_ruler = {
			limit = {
				is_landed = yes
				afl_can_borrow_trigger = yes
				NOT = { has_character_flag = afl_test_requester }
			}
			save_event_target_as = afltest_b
			# out of debt first: a borrower 100 gold in debt (the Fatimid caliph, box run 2026-11-15)
			# left the first month's margin unpaid
			while = { count = 400 limit = { NOT = { wealth = 0 } } wealth = 25 }
			wealth = 100
		}
		random_independent_ruler = {
			limit = {
				is_landed = yes
				afl_can_borrow_trigger = yes
				NOT = { has_character_flag = afl_test_requester }
				NOT = { character = event_target:afltest_b }
			}
			save_event_target_as = afltest_b2
			# out of debt first: a borrower 100 gold in debt (the Fatimid caliph, box run 2026-11-15)
			# left the first month's margin unpaid
			while = { count = 400 limit = { NOT = { wealth = 0 } } wealth = 25 }
			wealth = 100
		}
		random_independent_ruler = {
			limit = {
				is_landed = yes
				afl_can_lend_trigger = yes
				check_variable = { which = afl_base value = 50 }
				NOT = { character = event_target:afltest_b }
				NOT = { character = event_target:afltest_b2 }
				NOT = { war_with = event_target:afltest_b }
				NOT = { war_with = event_target:afltest_b2 }
			}
			save_event_target_as = afltest_l
		}
		random_independent_ruler = {
			limit = {
				is_landed = yes
				war = no
				NOT = { afl_in_contract_trigger = yes }
				NOT = { character = event_target:afltest_l }
			}
			save_event_target_as = afltest_p
		}
		random_independent_ruler = {
			limit = {
				is_landed = yes
				NOT = { any_demesne_title = { afl_has_ships_trigger = yes } }
			}
			save_event_target_as = afltest_n
		}
		if = {
			limit = {
				OR = {
					NOT = { event_target:afltest_b = { always = yes } }
					NOT = { event_target:afltest_b2 = { always = yes } }
					NOT = { event_target:afltest_l = { always = yes } }
					NOT = { event_target:afltest_p = { always = yes } }
					NOT = { event_target:afltest_n = { always = yes } }
				}
			}
			log = "CK2TEST fail scenario setup: missing a ruler at war, one who can spare 50 galleys, one at peace or one without ships"
			log = "CK2TEST done"
		}
		if = {
			limit = {
				event_target:afltest_b = { always = yes }
				event_target:afltest_b2 = { always = yes }
				event_target:afltest_l = { always = yes }
				event_target:afltest_p = { always = yes }
				event_target:afltest_n = { always = yes }
			}
			log = "CK2TEST pass scenario setup"
			log = "afltest: scenario borrower [afltest_b.GetBestName] ([afltest_b.GetID]), second borrower [afltest_b2.GetBestName] ([afltest_b2.GetID]), lender [afltest_l.GetBestName] ([afltest_l.GetID]), at peace [afltest_p.GetBestName] ([afltest_p.GetID]), no ships [afltest_n.GetBestName] ([afltest_n.GetID])"
			event_target:afltest_l = {
				log = "afltest: the lender's own holdings give [This.afl_base.GetValue] galleys before bonuses"
				any_demesne_title = {
					limit = { afl_has_ships_trigger = yes }
					log = "afltest: lender's holding with ships: [This.GetName] ([This.GetID])"
				}
			}

			# The conditions (common/scripted_triggers/afl_triggers.txt).
			if = { limit = { event_target:afltest_p = { NOT = { afl_can_borrow_trigger = yes } } } log = "CK2TEST pass a ruler at peace may not borrow" }
			if = { limit = { NOT = { event_target:afltest_p = { NOT = { afl_can_borrow_trigger = yes } } } } log = "CK2TEST fail a ruler at peace may not borrow: [afltest_p.GetBestName] ([afltest_p.GetID]) may" }
			if = { limit = { event_target:afltest_b = { afl_can_borrow_trigger = yes } } log = "CK2TEST pass a ruler at war may borrow" }
			if = { limit = { NOT = { event_target:afltest_b = { afl_can_borrow_trigger = yes } } } log = "CK2TEST fail a ruler at war may borrow: [afltest_b.GetBestName] ([afltest_b.GetID]) may not" }
			if = { limit = { event_target:afltest_n = { NOT = { afl_can_lend_trigger = yes } } } log = "CK2TEST pass a ruler without a holding with ships may not lend" }
			if = { limit = { NOT = { event_target:afltest_n = { NOT = { afl_can_lend_trigger = yes } } } } log = "CK2TEST fail a ruler without a holding with ships may not lend: [afltest_n.GetBestName] ([afltest_n.GetID]) may" }
			if = { limit = { event_target:afltest_l = { afl_can_lend_trigger = yes } } log = "CK2TEST pass a ruler with ships may lend" }
			if = { limit = { NOT = { event_target:afltest_l = { afl_can_lend_trigger = yes } } } log = "CK2TEST fail a ruler with ships may lend: [afltest_l.GetBestName] ([afltest_l.GetID]) may not" }
			event_target:afltest_b = { set_character_flag = afl_request_pending }
			if = { limit = { event_target:afltest_b = { NOT = { afl_can_borrow_trigger = yes } } } log = "CK2TEST pass a pending request blocks another" }
			if = { limit = { NOT = { event_target:afltest_b = { NOT = { afl_can_borrow_trigger = yes } } } } log = "CK2TEST fail a pending request blocks another: [afltest_b.GetBestName] ([afltest_b.GetID]) may borrow" }
			event_target:afltest_b = { clr_character_flag = afl_request_pending }

			# The cut's arithmetic, on the ruler at peace (afl_count_cut_effect).
			event_target:afltest_p = { set_variable = { which = afl_ask value = 50 } set_variable = { which = afl_base value = 50 } afl_count_cut_effect = yes }
			if = { limit = { event_target:afltest_p = { is_variable_equal = { which = afl_k value = 100 } } } log = "CK2TEST pass 50 galleys of 50 cut each holding 100%" }
			if = { limit = { NOT = { event_target:afltest_p = { is_variable_equal = { which = afl_k value = 100 } } } } log = "CK2TEST fail 50 galleys of 50 cut each holding 100%: it gave [afltest_p.afl_k.GetValue]" }
			event_target:afltest_p = { set_variable = { which = afl_ask value = 50 } set_variable = { which = afl_base value = 73 } afl_count_cut_effect = yes }
			if = { limit = { event_target:afltest_p = { is_variable_equal = { which = afl_k value = 69 } } } log = "CK2TEST pass 50 galleys of 73 cut each holding 69%" }
			if = { limit = { NOT = { event_target:afltest_p = { is_variable_equal = { which = afl_k value = 69 } } } } log = "CK2TEST fail 50 galleys of 73 cut each holding 69%: it gave [afltest_p.afl_k.GetValue]" }
			event_target:afltest_p = { set_variable = { which = afl_ask value = 150 } set_variable = { which = afl_base value = 400 } afl_count_cut_effect = yes }
			if = { limit = { event_target:afltest_p = { is_variable_equal = { which = afl_k value = 38 } } } log = "CK2TEST pass 150 galleys of 400 cut each holding 38%" }
			if = { limit = { NOT = { event_target:afltest_p = { is_variable_equal = { which = afl_k value = 38 } } } } log = "CK2TEST fail 150 galleys of 400 cut each holding 38%: it gave [afltest_p.afl_k.GetValue]" }
			event_target:afltest_p = { set_variable = { which = afl_ask value = 100 } set_variable = { which = afl_base value = 300 } afl_count_cut_effect = yes }
			if = { limit = { event_target:afltest_p = { is_variable_equal = { which = afl_k value = 34 } } } log = "CK2TEST pass 100 galleys of 300 cut each holding 34%" }
			if = { limit = { NOT = { event_target:afltest_p = { is_variable_equal = { which = afl_k value = 34 } } } } log = "CK2TEST fail 100 galleys of 300 cut each holding 34%: it gave [afltest_p.afl_k.GetValue]" }
			event_target:afltest_p = { set_variable = { which = afl_ask value = 500 } set_variable = { which = afl_base value = 73 } afl_count_cut_effect = yes }
			if = { limit = { event_target:afltest_p = { is_variable_equal = { which = afl_k value = 100 } } } log = "CK2TEST pass 500 galleys of 73 cut each holding 100%" }

# afltest.13, near the year's end: up to three new 50-galley loans for the save checks, which
# read the year's last save for the lent fleets (save_checks.txt). Each needs a ruler at war and
# a free lender who can spare 50 and is not at war with it; a fleet whose war ends first disbands.
character_event = {
	id = afltest.13
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		any_independent_ruler = {
			limit = { is_landed = yes afl_can_lend_trigger = yes }
			afl_count_base_effect = yes
		}
		while = {
			count = 3
			limit = { always = yes }
			random_independent_ruler = {
				limit = {
					is_landed = yes
					afl_can_borrow_trigger = yes
					any_independent_ruler = {
						is_landed = yes
						afl_can_lend_trigger = yes
						check_variable = { which = afl_base value = 50 }
						NOT = { character = PREV }
						NOT = { war_with = PREV }
					}
				}
				save_event_target_as = afl_borrower
				set_character_flag = afl_request_pending
				set_variable = { which = afl_ask value = 50 }
				random_independent_ruler = {
					limit = {
						is_landed = yes
						afl_can_lend_trigger = yes
						check_variable = { which = afl_base value = 50 }
						NOT = { character = PREV }
						NOT = { war_with = PREV }
					}
					save_event_target_as = afl_lender
				}
				afl_set_terms_effect = yes
				event_target:afl_lender = { afl_start_war_loan_effect = yes }
				clr_character_flag = afl_request_pending
			}
		}
		if = {
			limit = { any_independent_ruler = { has_character_flag = afl_borrowing has_earmarked_regiments = afl_war_loan } }
			log = "CK2TEST pass a late loan runs into the year's last save"
		}
		if = {
			limit = { NOT = { any_independent_ruler = { has_character_flag = afl_borrowing has_earmarked_regiments = afl_war_loan } } }
			log = "CK2TEST fail a late loan runs into the year's last save: no pair of rulers could start one"
		}
		# A late charter too (afltest_charter_events.txt).
		character_event = { id = afltest.38 }
	}
}
