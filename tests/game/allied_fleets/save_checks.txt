# Checks on the save the campaign test's allied_fleets run ends with (tools/ck2save.py check).
# One per line: present NAME | absent NAME | count PATH OP N | every PATH OP VALUE; PATH parts
# may filter, KEY[SUBKEY=VALUE]. See tools/ck2save.py.
present afl_test_started                            # the tests' global flag: they ran
count character/*/flags/afl_test_requester >= 20   # the AI askers the tests picked
absent afl_request_pending                          # every request was answered
absent afl_request_incoming                         # and every lender is free of it
count character/*/dmn/navy/sub_unit[earmark=afl_war_loan] >= 1   # afltest.13's late loans reach the save
every character/*/dmn/navy/sub_unit[earmark=afl_war_loan]/troops/ga == 50   # each lent fleet is the 50 galleys asked for
count character/*/dmn/navy/sub_unit[earmark=afl_war_loan]/maintenance_multiplier == 0  # spawned at the usual wages: a multiplier of 1 writes no line (docs/ck2-notes.md)
absent afl_offer_out                                # every charter offer was answered
absent afl_offer_in
count character/*/dmn/navy/sub_unit[earmark=afl_charter] >= 1   # afltest.38's late charter sails before the save
every character/*/dmn/navy/sub_unit[earmark=afl_charter]/troops/ga == 50   # each chartered fleet is the 50 galleys asked for
count character/*/dmn/navy/sub_unit[earmark=afl_charter]/maintenance_multiplier == 0  # at the usual wages
